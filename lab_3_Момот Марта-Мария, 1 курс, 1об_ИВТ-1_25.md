    double w2;
    double wz;

    return a.w1 * b.w1 + a.w2 * b.w2+a.wz *b.wz;
}
SpaceVec getCross(SpaceVec a, SpaceVec b) {
    SpaceVec it;
    snprintf(it.te, sizeof(it.te), "it");
    it.w1 = a.w2*b.wz-a.wz*b.w2;
    it.w2 = a.wz * b.w1 - a.w1 * b.wz;
    it.wz = a.w1 * b.w2 - a.w2 * b.w1;
    return it;
}
double getMod(SpaceVec v) {
    return sqrt(v.w1 * v.w1 + v.w2 * v.w2 + v.wz * v.wz);
}
void printVec(SpaceVec v) {
    printf("%s: (%.2f, %.2f, %.2f)\n", v.te, v.w1, v.w2, v.wz);
}

    printVec(veks);
    printf("%.2f\n", getScalar(vek, veks));

Вычислить, используя структуру комплексного числа, комплексную экспоненту exp(z) некоторого z ∈ C: exp(z) = 1 + z + 1 2!z 2 + 1 3!z 3 + . . . + 1 n! z n .

Список идентификаторов


---------------------------------------------------------

|Имя переменной|Тип данных      |     Описание       |

------------------------------------------------


|mni     |    double                       |Поле структуры: мнимая часть|


------------------------------------------------

|cs|Complex            |Структура-введенное число|

------------------------------------------------

|it  |Complex            |Структура, в которой считается итог|

Код программы 
<span style="color:yellow">#include</span><span style="color:green"><stdio.h>< / span>
typedef struct {
    double ves; // Действительная часть (re)
    double mni; // Мнимая часть (im)
} Complex;
<span style="color:red">int</span> main() {
    Complex cs;
    Complex it = { 1.0, 0.0 };
    Complex st7 = { 1.0, 0.0 };
    int n;
    if (scanf("%lf %lf %d", &cs.ves, &cs.mni, &n) != 3) return 1;
    for (int ss = 1; ss <= n; ss++) {
        double old_st7_ves = st7.ves;
        st7.ves = (old_st7_ves * cs.ves - st7.mni * cs.mni) / ss;
        st7.mni = (old_st7_ves * cs.mni + st7.mni * cs.ves) / ss;
        it.ves += st7.ves;
        it.mni += st7.mni;
    }
    printf("%.4f + %.4fi\n", it.ves, it.mni);
    <span style="color:red">return 0</span>;
}

Постановка задачи
Создайте так называемое размеченное объединение union, которое заключено в виде поля структуры struct вместе с ещё одним полем, которое является перечислением enum и служит индикатором того, что именно на текущий момент хранится в таком вложенном объединении. Создать и заполнить динамический массив таких структур с объединениями внутри, заполняя вспомогательное поле перечисления enum для сохранения информации о хранимом в каждом размеченном объединении типе данных. Реализовать распечатку данных массива таких структур в консоль.


Список идентификаторов


--------------------------------------



|Имя переменной|Тип данных | Описание |

--------------------------------------


|is |enum |В объединении есть int|

--------------------------------------



|bm| TypeMarker|Поле структуры- активный тип данных|

--------------------------------------



|sj| int|Поле объединения- целое число|

--------------------------------------



|dro|float|Поле объединения: хранит дробное число |

--------------------------------------




|aq| union|Объединение для экономии памяти |


--------------------------------------



|inall|int|Количество элементов массива|

--------------------------------------




|qv|int|Ввод 0 для целого, 1 для дробного |

--------------------------------------

--------------------------------------


Код программы
<span style="color:yellow">#include</span><span style="color:green"><stdio.h>< / span>
<span style="color:yellow">#include</span><span style="color:green"> <stdlib.h>< / span>

typedef enum {
    is,
    qq
} TypeMarker;

typedef struct {
    TypeMarker bm;
    union {
        int sj;
        float dro;
    } aq;
} Container;
<span style="color:red">int</span>main() {
    int inall;
    if (scanf_s("%d", &inall) != 1) return 0;
    Container* dw = (Container*)malloc(inall * sizeof(Container));
    if (dw == NULL) return 1;
    for (int ql = 0; ql < inall; ql++) {
        int qv;
        if (scanf_s("%d", &qv) != 1) break;

        if (qv == 0) {
            dw[ql].bm = is;
            scanf_s("%d", &dw[ql].aq.sj);
        }
        else {
            dw[ql].bm = qq;
            scanf_s("%f", &dw[ql].aq.dro);
        }
    }
    for (int ql = 0; ql < inall; ql++) {
        if (dw[ql].bm == is) {
            printf("%d\n", dw[ql].aq.sj);
        }
        else {
            printf("%.2f\n", dw[ql].aq.dro);
        }
    }
    free(dw);
    <span style="color:red">return 0</span>;
}
