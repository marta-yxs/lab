        for(j = 0; j < cols; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }
    for(i = 0; i < rows; i++) {
        int sum = 0;
        for(j = 0; j < cols; j++) {
            sum = sum + matrix[i][j];
        }
        avg = (double)sum / cols;
        matrix[i][0] = (int)avg;
    }
    printf("\nMatrix after replacement:\n");
    for(i = 0; i < rows; i++) {
        for(j = 0; j < cols; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }

    <font color="green">return 0</font>;
}

#include <stdio.h>

<font color="red">int</font> main() {
    int n;
    int i, j;
    int key;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];

    for(i = 0; i < n; i++) {
        printf("arr[%d] = ", i);

    for(i = 1; i < n; i++) {
        key = arr[i];
        j = i - 1;
        while(j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j = j - 1;
        }

        arr[j + 1] = key;
    }

        printf("%d ", arr[i]);
